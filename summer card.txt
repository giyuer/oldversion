local function readConfig()
    local ok, raw = pcall(function() if isfile and isfile(CONFIG_PATH) then return readfile(CONFIG_PATH) end end)
    if ok and raw then
        local ok2, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if ok2 and type(data) == "table" then
            state.autoSmart = data.autoSmart == true
            state.autoPriority = data.autoPriority == true
            state.voteDelay = tonumber(data.voteDelay) or 1.0
            state.priority = type(data.priority) == "table" and data.priority or {}
        end
    end
end

local function saveConfig()
    pcall(function()
        if not writefile then return end
        if makefolder and (not isfolder or not isfolder("MarusHub")) then pcall(makefolder, "MarusHub") end
        writefile(CONFIG_PATH, HttpService:JSONEncode({ autoSmart = state.autoSmart, autoPriority = state.autoPriority, voteDelay = state.voteDelay, priority = state.priority }))
    end)
end

-- DETECT ENGINE
local function getModeCards()
    local gameUI = PlayerGui:FindFirstChild("GameUI")
    local mc = gameUI and gameUI:FindFirstChild("ModeCards")
    local ps = mc and mc:FindFirstChild("PathSelection")
    return mc, ps and ps:FindFirstChild("Cards")
end

local function collectCards(cardsFrame)
    local list = {}
    for _, c in ipairs(cardsFrame:GetChildren()) do
        if c:IsA("GuiObject") and not c:IsA("UILayout") then
            local titleLabel = c:FindFirstChild("Title", true)
            local hitbox = c:FindFirstChild("Hitbox", true)
            if titleLabel and hitbox and titleLabel:IsA("TextLabel") then
                local name = tostring(titleLabel.Text):gsub("^%s+", ""):gsub("%s+$", "")
                local textCollected = {}
                for _, obj in ipairs(c:GetDescendants()) do
                    if obj:IsA("TextLabel") then
                        local rawText = tostring(obj.Text):gsub("^%s+", ""):gsub("%s+$", "")
                        if rawText ~= "" and obj ~= titleLabel and not tonumber(rawText) and rawText ~= name and obj.Name ~= "Order" then table.insert(textCollected, rawText) end
                    end
                end
                if name ~= "" then table.insert(list, { name = name, buff = table.concat(textCollected, " "):lower(), hitbox = hitbox, order = c.LayoutOrder }) end
            end
        end
    end
    table.sort(list, function(a, b) return a.order < b.order end) return list
end

local function chooseCard(cards)
    if state.autoPriority then
        local bestRank, best
        for _, card in ipairs(cards) do
            for rank, wantedName in ipairs(state.priority) do
                if card.name:lower() == wantedName:lower() then if not bestRank or rank < bestRank then bestRank = rank best = card end break end
            end
        end
        if best then return best, "Custom List" end
    end
    if state.autoSmart then
        for _, wantedKeyword in ipairs(SMART_KEYWORDS) do
            for _, card in ipairs(cards) do if string.find(card.buff, wantedKeyword) then return card, "Smart Mode (" .. wantedKeyword .. ")" end end
        end
    end
    return cards[1], "Default (Card 1)"
end

local currentSignature, votedSignature = nil, nil
task.spawn(function()
    while not unloaded do
        task.wait(0.35)
        local mc, cardsFrame = getModeCards()
        if not mc or not cardsFrame or not mc.Enabled then if currentSignature ~= nil then currentSignature = nil setStatus("Idle", COL_DIM) end continue end
        local cards = collectCards(cardsFrame)
        if #cards == 0 then continue end

        local signature = ""
        for _, c in ipairs(cards) do signature = signature .. c.name .. "|" end
        if signature ~= currentSignature then currentSignature = signature setStatus("New cards detected (" .. #cards .. ")", COL_ACCENT) end

        if signature ~= votedSignature then
            local chosen, reason = chooseCard(cards)
            if chosen then
                task.wait(state.voteDelay)
                if mc.Enabled and chosen.hitbox and firesignal then
                    firesignal(chosen.hitbox.Activated) votedSignature = signature
                    setStatus("Voted: " .. chosen.name .. " [" .. reason .. "]", COL_ON)
                end
            end
        end
    end
end)
